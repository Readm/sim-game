292  virtual void tick() {
293  #if SIM_BUILD_MODE == SIM_DEBUG_MODE
294  auto pre_tick_state = serialize();
295  #endif
296 
297  // 先调用所有端口的tick
298  for (auto& [name, port] : p_state_.input_ports) {
299  port->tick();
300  }
301  for (auto& [name, port] : p_state_.output_ports) {
302  port->tick();
303  }
304 
305  onTick();
306 
307  if (parallelization_method_ == ParallelizationMethod::THREAD_POOL && thread_pool_) {
308  std::vector<std::future<void>> futures;
309  for (auto& child : p_state_.children) {
310  futures.push_back(thread_pool_->enqueue([&child] {
311  child->tick();
312  }));
313  }
314  // 等待所有子节点完成tick
315  for (auto& future : futures) {
316  future.wait();
317  }
318  } else {
319  // 串行执行
320  for (auto& child : p_state_.children) {
321  child->tick();
322  }
323  }
324 
325  #if SIM_BUILD_MODE == SIM_DEBUG_MODE
326  auto post_tick_state = serialize();
327  if (pre_tick_state != post_tick_state) {
328  SIM_WARNING("Node state changed during tick operation");
329  }
330  #endif
331  }
