96constexpr TypeID generateTypeID(const char* name) {
97 // 简单的字符串哈希函数
98 TypeID hash = 5381;
99 while (*name) {
100 hash = ((hash << 5) + hash) + static_cast<unsigned char>(*name);
101 name++;
102 }
103 return hash;
104}
