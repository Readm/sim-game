31  for(size_t i = 0; i < num_threads; ++i) {
32  workers_.emplace_back([this] {
33  while(true) {
34  std::function<void()> task;
35  {
36  std::unique_lock<std::mutex> lock(queue_mutex_);
37  condition_.wait(lock, [this] {
38  return stop_ || !tasks_.empty();
39  });
40

42  return;
43  }

47  }

61  );
62

64  {

66  if(stop_) {

69  tasks_.emplace([task](){ (*task)(); });
70  }
